create function fn_calcular_total_pedido(p_id_pedido int)
returns decimal(10,2)
deterministic
begin
declare subtotal_pizzas decimal(10,2);
declare envio decimal(10,2);
declare total_final decimal(10,2);
set subtotal_pizzas = 0.00;
set envio = 0.00;
set total_final = 0.00;
select case when sum(cantidad * precio_unitario) is not null then sum(cantidad * precio_unitario)
else 0.00 
end into subtotal_pizzas 
from detalle_pedidos where id_pedido = p_id_pedido;
if exists (select 1 from domicilios where id_pedido = p_id_pedido) then select costo_envio into envio from domicilios where id_pedido = p_id_pedido;
end if;
set total_final = (subtotal_pizzas + envio) * 1.19;
return total_final;
end //
delimiter ;



delimiter //
create function fn_ganancia_neta_diaria(p_fecha date)
returns decimal(10,2)
deterministic
begin
declare total_ventas decimal(10,2);
declare total_costos decimal(10,2);
select sum(total) into total_ventas from pedidos where date(fecha_hora) = p_fecha;
select sum(dp.cantidad * pi.cantidad_requerida * i.costo_unitario) into total_costos from pedidos p
join detalle_pedidos dp on p.id_pedido = dp.id_pedido
join pizza_ingredientes pi on dp.id_pizza = pi.id_pizza
join ingredientes i on pi.id_ingrediente = i.id_ingrediente where date(p.fecha_hora) = p_fecha;
return total_ventas - total_costos;
end //
delimiter ;



delimiter //
create procedure sp_registrar_entrega_domicilio(in p_id_domicilio int, in p_hora_entrega datetime)
begin
update domicilios
set hora_entrega = p_hora_entrega where id_domicilio = p_id_domicilio;
update pedidos
set estado = 'entregado' where id_pedido = (select id_pedido from domicilios  where id_domicilio = p_id_domicilio);
end //
delimiter ;
