select distinct c.id_cliente, c.nombre, c.telefono, c.email from clientes c join pedidos p on c.id_cliente = p.id_cliente
where p.fecha_hora between '2026-08-01 00:00:00' and '2026-08-31 23:59:59';


select p.nombre as pizza, p.tamano, count(dp.id_pizza) as veces_pedida, sum(dp.cantidad) as total_unidades_vendidas from detalle_pedidos dp
join pizzas p on dp.id_pizza = p.id_pizza group by p.id_pizza, p.nombre, p.tamano order by total_unidades_vendidas desc;


select r.nombre as repartidor,
r.zona_asignada, p.id_pedido, p.fecha_hora, p.estado as estado_pedido from domicilios d join repartidores r on d.id_repartidor = r.id_repartidor
join pedidos p on d.id_pedido = p.id_pedido;


select r.zona_asignada, avg(time_to_sec(timediff(d.hora_entrega, d.hora_salida)) / 60) as promedio_minutos
from domicilios d join repartidores r on d.id_repartidor = r.id_repartidor where d.hora_entrega is not null group by r.zona_asignada;


select c.id_cliente, c.nombre as cliente, sum(p.total) as total_gastado from clientes c join pedidos p on c.id_cliente = p.id_cliente
group by c.id_cliente, c.nombre having sum(p.total) > 50000;


select id_pizza, nombre, tamano, precio_base, tipo from pizzas where nombre like '%hawaiana%';



select id_cliente, nombre, telefono, email from clientes where id_cliente in (select id_cliente from pedidos where month(fecha_hora) = month(current_date()) 
and year(fecha_hora) = year(current_date()) group by id_cliente having count(id_pedido) > 5);
